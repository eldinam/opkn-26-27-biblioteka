02
Dio drugi

Struktura projekta

Put zahtjeva

Kroz koje fajlove prolazi GET /

Pokretanje

package.json i bin/www

package.json
"scripts": {
  "start": "node ./bin/www"
}

npm start je samo prečica za ovu komandu.

bin/www
var app = require('../app');
var http = require('http');

var port = normalizePort(
  process.env.PORT || '3000');

var server = http.createServer(app);
server.listen(port);

Ovdje je broj porta. Probajte '3001'.

Srce aplikacije

app.js

// 1. paketi i naši fajlovi
var express = require('express');
var indexRouter = require('./routes/index');
var app = express();

// 2. šabloni su u views/ i pišu se u EJS-u
app.set('views', path.join(__dirname, 'views'));
app.set('view engine', 'ejs');

// 3. šta se radi sa svakim zahtjevom
app.use(logger('dev'));
app.use(express.static(path.join(__dirname, 'public')));

// 4. koja adresa ide u koju rutu
app.use('/', indexRouter);

// 5. ništa nije odgovaralo: 404
app.use(function(req, res, next) {
  next(createError(404));
});

Čita se odozgo prema dolje.

Zahtjev prolazi kroz svaki app.use redom, dok ga neko ne obradi.

Ruta i šablon

Podaci idu iz rute u šablon

routes/index.js
router.get('/', function (req, res) {
  res.render('index', {
    title: 'Biblioteka PMF'
  });
});
views/index.ejs
<h1><%= title %></h1>
u browseru
<h1>Biblioteka PMF</h1>
'index' u res.render je ime fajla views/index.ejs. Svaki ključ iz objekta postaje varijabla u šablonu.
Moduli

require i module.exports

routes/index.js
const router = express.Router();
// ...
module.exports = router;   // izvoz
app.js
var indexRouter = require('./routes/index');  // uvoz

Korak 3 i 4

Ruta šalje, šablon prikazuje

routes/index.js
const dayjs = require('dayjs');
require('dayjs/locale/bs');
const zanimljivosti = require('../data/zanimljivosti');
dayjs.locale('bs');

router.get('/', function (req, res, next) {
  const indeks = Math.floor(
    Math.random() * zanimljivosti.length);

  res.render('index', {
    title: 'Biblioteka PMF',
    datum: dayjs().format('dddd, D. MMMM YYYY.'),
    zanimljivost: zanimljivosti[indeks],
  });
});
views/index.ejs
<h1><%= title %></h1>
<p class="datum"><%= datum %></p>

<section class="kartica">
  <h2>Zanimljivost dana</h2>
  <p><%= zanimljivost %></p>
</section>

Isti ključevi u ruti i u šablonu: title, datum, zanimljivost.
