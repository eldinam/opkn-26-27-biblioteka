Prve rute

Prva ruta

Novi fajl i dvije linije u app.js

routes/knjige.js
const express = require("express");
const knjige = require("../data/knjige");

const router = express.Router();

router.get("/", function (req, res) {
  res.json(knjige);
});

module.exports = router;
app.js
var knjigeRouter = require('./routes/knjige');

// ...
app.use('/knjige', knjigeRouter);

"/" u routeru znači „tačno /knjige“, jer je /knjige već dodan u app.js.

Statične rute

Putanja je uvijek ista

// GET /knjige/broj
router.get("/broj", function (req, res) {
  res.json({ broj: knjige.length });
});

// GET /knjige/nasumicna
router.get("/nasumicna", function (req, res) {
  const indeks = Math.floor(Math.random() * knjige.length);
  res.json(knjige[indeks]);
});
Dinamičke rute

Dio putanje je varijabla

// GET /knjige/3, /knjige/5, /knjige/99 ...
router.get("/:id", function (req, res) {
  const id = Number(req.params.id);
  const knjiga = knjige.find((k) => k.id === id);

  if (!knjiga) {
    return res.status(404).json({
      greska: `Knjiga ${req.params.id} ne postoji` });
  }
  res.json(knjiga);
});

Kod

Ruta /knjige/naslov/:slug

const slugify = require("slugify");

function napraviSlug(naslov) {
  return slugify(naslov, { lower: true, strict: true });
}

// GET /knjige/naslov/dervis-i-smrt
router.get("/naslov/:slug", function (req, res) {
  const knjiga = knjige.find(
    (k) => napraviSlug(k.naslov) === req.params.slug);

  if (!knjiga) {
    return res.status(404).json({ greska: "Nema" });
  }
  res.json(knjiga);
});
> npm install slugify
  1. iz adrese: "dervis-i-smrt"
  2. za svaku knjigu napravi slug naslova
  3. vrati onu čiji se slug poklapa
